Location: client/index.html:31
Severity: High

// Add authentication + ownership check at the route level:
app.patch("/api/assessments/:id", async (req: any, res) => {
  // ... existing sanitization ...

  const assessment = await storage.getAssessmentById(req.params.id);
  if (!assessment) return res.status(404).json({ message: "Not found" });

  // Ownership check
  if (assessment.userId) {
    if (!req.isAuthenticated() || req.user.userId !== assessment.userId) {
      return res.status(403).json({ message: "Forbidden" });
    }
  } else {
    const guestToken = req.cookies?.guest_token;
    if (!guestToken || guestToken !== assessment.guestSessionId) {
      return res.status(403).json({ message: "Forbidden" });
    }
  }
  // ... rest of handler
