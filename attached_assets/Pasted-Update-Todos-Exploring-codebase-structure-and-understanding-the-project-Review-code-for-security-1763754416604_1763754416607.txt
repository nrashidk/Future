// server/middleware/orgAdmin.middleware.ts (NEW FILE)
export const isOrgAdmin = async (req: any, res: any, next: any) => {
  if (!req.user) {
    return res.status(401).json({ message: "Unauthorized" });
  }
  
  const userId = req.user.isLocal ? req.user.userId : req.user.claims.sub;
  
  try {
    const user = await storage.getUser(userId);
    
    if (!user) {
      return res.status(401).json({ message: "User not found" });
    }
    
    // Allow both superadmin and org_admin
    if (user.role === "superadmin" || user.accountType === "org_admin") {
      req.userId = userId;
      req.user accountType = user.accountType;
      return next();
    }
    
    return res.status(403).json({ message: "Forbidden: Admin access required" });
  } catch (error) {
    res.status(500).json({ message: "Authorization check failed" });
  }
};
